import {
  FinancialExtractedData,
  FinancialInconsistency,
  FinancialIntakeState,
} from "./types";

export type FinancialConfirmationStatus =
  | "not_required"
  | "pending"
  | "confirmed"
  | "rejected";

export interface FinancialConfirmationItem {
  field: string;
  value: unknown;
  previousValue?: unknown;
  reason: string;
  severity: FinancialInconsistency["severity"];
  status: FinancialConfirmationStatus;
}

export interface FinancialConfirmationResult {
  status: FinancialConfirmationStatus;
  confirmedData: FinancialExtractedData[];
  rejectedFields: string[];
  pendingItems: FinancialConfirmationItem[];
}

export class FinancialHumanConfirmation {
  getPendingItems(
    state: FinancialIntakeState
  ): FinancialConfirmationItem[] {
    const items: FinancialConfirmationItem[] = [];

    for (const inconsistency of state.inconsistencies) {
      if (!inconsistency.requiresConfirmation) {
        continue;
      }

      for (const field of inconsistency.fields) {
        const values = state.extractedData.filter(
          (data) => data.field === field
        );

        const latest = values[values.length - 1];
        const previous =
          values.length > 1
            ? values[values.length - 2]
            : undefined;

        items.push({
          field,
          value: latest?.value,
          previousValue: previous?.value,
          reason: inconsistency.description,
          severity: inconsistency.severity,
          status: "pending",
        });
      }
    }

    return items;
  }

  requiresConfirmation(
    state: FinancialIntakeState
  ): boolean {
    return state.inconsistencies.some(
      (item) => item.requiresConfirmation
    );
  }

  buildConfirmationQuestion(
    item: FinancialConfirmationItem
  ): string {
    if (item.previousValue !== undefined) {
      return (
        `DetectÃ© una diferencia en "${item.field}". ` +
        `El valor anterior era ${this.formatValue(
          item.previousValue
        )} y el nuevo valor es ${this.formatValue(
          item.value
        )}. ` +
        `Â¿CuÃ¡l de los dos valores debemos utilizar?`
      );
    }

    return (
      `Necesito confirmar el dato "${item.field}" ` +
      `con valor ${this.formatValue(item.value)}. ` +
      `Â¿Confirmas que este dato es correcto?`
    );
  }

  confirm(
    state: FinancialIntakeState,
    field: string,
    use: "latest" | "previous" = "latest"
  ): FinancialConfirmationResult {
    const item = this.getPendingItems(state).find(
      (pending) => pending.field === field
    );

    if (!item) {
      return {
        status: "not_required",
        confirmedData: [],
        rejectedFields: [],
        pendingItems: [],
      };
    }

    const values = state.extractedData.filter(
      (data) => data.field === field
    );

    const selected =
      use === "previous"
        ? values[values.length - 2]
        : values[values.length - 1];

    if (!selected) {
      return {
        status: "not_required",
        confirmedData: [],
        rejectedFields: [],
        pendingItems: this.getPendingItems(state),
      };
    }

    const confirmedData: FinancialExtractedData[] = [
      {
        ...selected,
        confirmed: true,
      },
    ];

    const remaining = this.getPendingItems(state).filter(
      (pending) => pending.field !== field
    );

    return {
      status:
        remaining.length > 0
          ? "pending"
          : "confirmed",
      confirmedData,
      rejectedFields: [],
      pendingItems: remaining,
    };
  }

  reject(
    state: FinancialIntakeState,
    field: string
  ): FinancialConfirmationResult {
    const item = this.getPendingItems(state).find(
      (pending) => pending.field === field
    );

    if (!item) {
      return {
        status: "not_required",
        confirmedData: [],
        rejectedFields: [],
        pendingItems: [],
      };
    }

    const pendingItems = this.getPendingItems(state).filter(
      (pending) => pending.field !== field
    );

    return {
      status:
        pendingItems.length > 0
          ? "pending"
          : "rejected",
      confirmedData: [],
      rejectedFields: [field],
      pendingItems,
    };
  }

  applyConfirmation(
    state: FinancialIntakeState,
    field: string,
    confirmed: boolean,
    use: "latest" | "previous" = "latest"
  ): FinancialIntakeState {
    const values = state.extractedData.filter(
      (data) => data.field === field
    );

    const selectedIndex =
      use === "previous"
        ? values.length - 2
        : values.length - 1;

    const selected = values[selectedIndex];

    if (!selected) {
      return state;
    }

    const selectedValue = selected.value;

    const updatedAnswers = state.answers
      .filter((answer) => answer.field !== field)
      .concat(
        state.answers
          .filter((answer) => answer.field === field)
          .slice(
            use === "previous" ? -2 : -1,
            use === "previous" ? -1 : undefined
          )
          .map((answer) => ({
            ...answer,
            confirmed,
          }))
      );

    const updatedExtractedData =
      state.extractedData
        .filter((data) => data.field !== field)
        .concat({
          ...selected,
          confirmed,
        });

    const updatedInconsistencies =
      confirmed
        ? state.inconsistencies.filter(
            (item) =>
              !item.fields.includes(field)
          )
        : state.inconsistencies;

    const hasPendingConfirmation =
      updatedInconsistencies.some(
        (item) => item.requiresConfirmation
      );

    const hasRejectedField =
      !confirmed &&
      state.inconsistencies.some(
        (item) => item.fields.includes(field)
      );

    return {
      ...state,
      answers: updatedAnswers,
      extractedData: updatedExtractedData,
      inconsistencies: updatedInconsistencies,
      status: hasPendingConfirmation
        ? "needs_confirmation"
        : hasRejectedField
          ? "blocked"
          : state.status,
      canContinueToAnalysis:
        confirmed &&
        !hasPendingConfirmation &&
        state.missingFields.length === 0,
    };
  }

  private formatValue(value: unknown): string {
    if (value === undefined || value === null) {
      return "sin valor";
    }

    if (typeof value === "number") {
      return value.toLocaleString("es-MX");
    }

    if (typeof value === "boolean") {
      return value ? "SÃ­" : "No";
    }

    return String(value);
  }
}

export class FinancialDataMapper {
  map(
    state: FinancialIntakeState
  ): FinancialDataMappingResult {
    const mappedData: FinancialMappedField[] = [];
    const unmappedFields: string[] = [];

    for (const data of state.extractedData) {
      if (!this.isSupportedField(data.field)) {
        unmappedFields.push(data.field);
        continue;
      }

      mappedData.push({
        field: data.field,
        value: data.value,
        source: data.source,
        confidence: data.confidence,
        confirmed: data.confirmed,
      });
    }

    return {
      mappedData,
      confirmedData: mappedData.filter(
        (item) => item.confirmed
      ),
      pendingConfirmation: mappedData.filter(
        (item) => !item.confirmed
      ),
      unmappedFields: [
        ...new Set(unmappedFields),
      ],
    };
  }

  mapConfirmedData(
    data: FinancialExtractedData[]
  ): FinancialMappedField[] {
    return data
      .filter(
        (item) =>
          item.confirmed &&
          this.isSupportedField(item.field)
      )
      .map((item) => ({
        field: item.field,
        value: item.value,
        source: item.source,
        confidence: item.confidence,
        confirmed: true,
      }));
  }

  mapToTransactions(
    input: FinancialTransactionMappingInput
  ): FinancialTransactionMappingResult {
    const mapped = this.map(input.state);

    const transactions: FinancialTransaction[] = [];

    for (const item of mapped.confirmedData) {
      const accountConfig =
        FIELD_ACCOUNT_CONFIG[item.field];

      if (!accountConfig) {
        continue;
      }

      const account = this.findAccount(
        input.accounts,
        accountConfig.code
      );

      if (!account?.id) {
        continue;
      }

      const amount = this.toNumber(item.value);

      if (amount === undefined) {
        continue;
      }

      transactions.push({
        organizationId:
          input.context.organizationId,

        workspaceId:
          input.context.workspaceId,

        accountId: account.id,

        periodId: input.period.id,

        date: input.period.endDate,

        description:
          this.buildTransactionDescription(
            item.field
          ),

        amount,

        currency: account.currency,

        type: account.type,

        source: "manual",

        sourceType: "internal",

        classification: "actual",
      });
    }

    return {
      transactions,
      unmappedFields: mapped.unmappedFields,
      pendingConfirmation:
        mapped.pendingConfirmation,
    };
  }

  getAccountConfiguration(
    field: string
  ):
    | {
        code: string;
        name: string;
        type: FinancialAccount["type"];
      }
    | undefined {
    return FIELD_ACCOUNT_CONFIG[field];
  }

  getValue(
    state: FinancialIntakeState,
    field: string
  ): unknown {
    const item = [...state.extractedData]
      .reverse()
      .find(
        (data) =>
          data.field === field &&
          data.confirmed
      );

    return item?.value;
  }

  hasConfirmedField(
    state: FinancialIntakeState,
    field: string
  ): boolean {
    return state.extractedData.some(
      (data) =>
        data.field === field &&
        data.confirmed
    );
  }

  private findAccount(
    accounts: FinancialAccount[],
    code: string
  ): FinancialAccount | undefined {
    return accounts.find(
      (account) =>
        account.code === code &&
        account.active
    );
  }

  ): number | undefined {
    if (typeof value === "number") {

    }

    if (typeof value !== "string") {

      .replace(/MXN/gi, "")
      .replace(/USD/gi, "")
      .replace(/,/g, "")
      .trim();

      ? number
      : undefined;

export class FinancialIntakeEngine {
  initialize(
    mode: FinancialIntakeState["mode"]
  ): FinancialIntakeState {
    const questions = this.buildInitialQuestions(mode);

    return {
      status: "in_progress",
      informationStatus: "insufficient",
      mode,
      currentQuestion: this.selectNextQuestion(
        questions,
        [],
        []
      ),
      questions,
      answers: [],
      extractedData: [],
      missingFields: questions
        .filter((question) => question.required)
        .map((question) => question.field),
      inconsistencies: [],
      completionPercentage: 0,
      canContinueToAnalysis: false,
    };
  }

  evaluate(
    request: FinancialIntakeRequest
  ): FinancialIntakeResponse {
    const state =
      request.state ??
      this.initialize(request.mode);

    const missingFields =
      this.calculateMissingFields(state);

    const completionPercentage =
      this.calculateCompletionPercentage(state);

    const informationStatus =
      this.determineInformationStatus(
        state,
        missingFields
      );

    const requiresConfirmation =
      state.inconsistencies.some(
        (item) => item.requiresConfirmation
      );

    const canContinueToAnalysis =
      informationStatus === "sufficient" &&
      !requiresConfirmation;

    const status =
      requiresConfirmation
        ? "needs_confirmation"
        : canContinueToAnalysis
          ? "complete"
          : "in_progress";

    const nextQuestion =
      canContinueToAnalysis
        ? undefined
        : this.selectNextQuestion(
            state.questions,
            state.answers,
            state.inconsistencies
          );

    return {
      status,
      informationStatus,
      nextQuestion,
      extractedData: state.extractedData,
      missingFields,
      inconsistencies: state.inconsistencies,
      requiresConfirmation,
      canContinueToAnalysis,
      completionPercentage,
    };
  }

  private selectNextQuestion(
    questions: FinancialIntakeQuestion[],
    answers: FinancialIntakeState["answers"],
    inconsistencies: FinancialIntakeState["inconsistencies"]
  ): FinancialIntakeQuestion | undefined {
    const answeredFields = new Set(
      answers
        .filter(
          (answer) =>
            answer.interpretedValue !== undefined
        )
        .map((answer) => answer.field)
    );

    const inconsistentFields = new Set(
      inconsistencies.flatMap(
        (item) => item.fields
      )
    );

    return [...questions]
      .filter(
        (question) =>
          !answeredFields.has(question.field) ||
          inconsistentFields.has(question.field)
      )
      .sort(
        (a, b) =>
          QUESTION_PRIORITY_ORDER[a.priority] -
          QUESTION_PRIORITY_ORDER[b.priority]
      )[0];
  }

  private calculateMissingFields(
    state: FinancialIntakeState
  ): string[] {
    const answeredFields = new Set(
      state.answers
        .filter(
          (answer) =>
            answer.interpretedValue !== undefined
        )
        .map((answer) => answer.field)
    );

    const inconsistentFields = new Set(
      state.inconsistencies.flatMap(
        (item) => item.fields
      )
    );

    return state.questions
      .filter(
        (question) =>
          question.required &&
          (!answeredFields.has(question.field) ||
            inconsistentFields.has(question.field))
      )
      .map((question) => question.field);
  }

  private calculateCompletionPercentage(
    state: FinancialIntakeState
  ): number {
    const requiredQuestions =
      state.questions.filter(
        (question) => question.required
      );

    if (requiredQuestions.length === 0) {
      return 100;
    }

    const answeredRequired =
      requiredQuestions.filter((question) =>
        state.answers.some(
          (answer) =>
            answer.field === question.field &&
            answer.interpretedValue !== undefined &&
            !state.inconsistencies.some(
              (item) =>
                item.fields.includes(
                  question.field
                ) &&
                item.requiresConfirmation
            )
        )
      );

    return Math.round(
      (answeredRequired.length /
        requiredQuestions.length) *
        100
    );
  }

  private determineInformationStatus(
    state: FinancialIntakeState,
    missingFields: string[]
  ): FinancialInformationStatus {
    if (state.inconsistencies.length > 0) {
      return "inconsistent";
    }

    if (missingFields.length > 0) {
      return "insufficient";
    }

    return "sufficient";
  }

import {

const PRIORITY_WEIGHT: Record<
  FinancialIntakeQuestion["priority"],
  number
> = {
  critical: 100,
  high: 75,
  medium: 50,
  low: 25,
};

export class FinancialQuestionEngine {
  getNextQuestion(
    state: FinancialIntakeState
  ): FinancialIntakeQuestion | undefined {
    const answeredFields = new Set(
      state.answers
        .filter((answer) => answer.confirmed !== false)
        .map((answer) => answer.field)
    );

    const inconsistentFields = new Set(
      state.inconsistencies.flatMap(
        (item) => item.fields
      )
    );

    const candidates = state.questions
      .filter((question) => {
        const answered = answeredFields.has(
          question.field
        );

        const inconsistent =
          inconsistentFields.has(question.field);

        return !answered || inconsistent;
      })
      .map((question) => ({
        question,
        score: this.calculatePriority(
          question,
          inconsistentFields.has(question.field)
        ),
      }))
      .sort((a, b) => b.score - a.score);

    return candidates[0]?.question;
  }

  getPendingQuestions(
    state: FinancialIntakeState
  ): FinancialIntakeQuestion[] {
    const answeredFields = new Set(
      state.answers
        .filter((answer) => answer.confirmed !== false)
        .map((answer) => answer.field)
    );

    const inconsistentFields = new Set(
      state.inconsistencies.flatMap(
        (item) => item.fields
      )
    );

    return state.questions
      .filter((question) => {
        const answered = answeredFields.has(
          question.field
        );

        const inconsistent =
          inconsistentFields.has(question.field);

        return !answered || inconsistent;
      })
      .sort(
        (a, b) =>
          this.calculatePriority(
            b,
            inconsistentFields.has(b.field)
          ) -
          this.calculatePriority(
            a,
            inconsistentFields.has(a.field)
          )
      );
  }

  shouldAskFollowUp(
    question: FinancialIntakeQuestion,
    state: FinancialIntakeState
  ): boolean {
    const relatedInconsistency =
      state.inconsistencies.some((item) =>
        item.fields.includes(question.field)
      );

    return relatedInconsistency;
  }

  private calculatePriority(
    question: FinancialIntakeQuestion,
    inconsistent: boolean
  ): number {
    let score = PRIORITY_WEIGHT[question.priority];

    if (question.required) {
      score += 20;
    }

    if (inconsistent) {
      score += 50;
    }

    return score;
  }
}

  async process(
    request: FinancialIntakeRequest
  ): Promise<FinancialIntakeProcessResult> {
    let state =
      request.state ??
      this.initialize(request.mode);

    let interview:
      | FinancialInterviewResult
      | undefined;

    if (
      request.message &&
      state.currentQuestion
    ) {
      interview =
        financialAIInterviewer.interpretAnswer(
          state.currentQuestion,
          request.message
        );

      const inconsistencies =
        financialAIInterviewer.validateAgainstState(
          interview,
          state
        );

      state = {
        ...state,
        answers: [
          ...state.answers,
          interview.answer,
        ],
        extractedData: [
          ...state.extractedData,
          ...interview.extractedData,
        ],
        inconsistencies: [
          ...state.inconsistencies,
          ...inconsistencies,
          ...interview.inconsistencies,
        ],
      };
    }

    const evaluation =
      financialIntakeEngine.evaluate({
        ...request,
        state,
      });

    state = {
      ...state,
      status: evaluation.status,
      informationStatus:
        evaluation.informationStatus,
      currentQuestion:
        evaluation.nextQuestion,
      missingFields:
        evaluation.missingFields,
      inconsistencies:
        evaluation.inconsistencies,
      completionPercentage:
        evaluation.completionPercentage,
      canContinueToAnalysis:
        evaluation.canContinueToAnalysis,
    };

    const confirmationRequired =
      financialHumanConfirmation.requiresConfirmation(
        state
      );

    if (confirmationRequired) {
      state = {
        ...state,
        status: "needs_confirmation",
        canContinueToAnalysis: false,
      };
    }

      financialDataMapper.map(state);

      status: state.status,

        state.informationStatus,
      nextQuestion:
        state.currentQuestion,
      extractedData:
        state.extractedData,
      missingFields:
        state.missingFields,
      inconsistencies:
        state.inconsistencies,
      requiresConfirmation:
        confirmationRequired,
      canContinueToAnalysis:
        state.canContinueToAnalysis &&
        !confirmationRequired,
      completionPercentage:
        state.completionPercentage,
      state,
      interview,
      mapping,
    };
  }

import {

  FinancialInconsistency,
  FinancialIntakeAnswer,

export interface FinancialInterviewResult {
  answer: FinancialIntakeAnswer;
  extractedData: FinancialExtractedData[];
  inconsistencies: FinancialInconsistency[];
  clarificationRequired: boolean;
  clarificationQuestion?: string;
}

export class FinancialAIInterviewer {
  interpretAnswer(
    question: FinancialIntakeQuestion,
    rawAnswer: string
  ): FinancialInterviewResult {
    const normalizedAnswer = rawAnswer.trim();

    if (!normalizedAnswer) {
      return {
        answer: {
          questionId: question.id,
          field: question.field,
          rawAnswer,
          confirmed: false,
        },
        extractedData: [],
        inconsistencies: [],
        clarificationRequired: true,
        clarificationQuestion:
          "Necesito una respuesta para poder continuar. Â¿Puedes proporcionar ese dato?",
      };
    }

    const interpretedValue = this.parseValue(
      question,
      normalizedAnswer
    );

    if (interpretedValue === undefined) {
      return {
        answer: {
          questionId: question.id,
          field: question.field,
          rawAnswer,
          confirmed: false,
        },
        extractedData: [],
        inconsistencies: [],
        clarificationRequired: true,
        clarificationQuestion:
          this.buildClarificationQuestion(question),
      };
    }

    const confidence =
      this.calculateConfidence(
        question,
        normalizedAnswer,
        interpretedValue
      );

    const extractedData: FinancialExtractedData[] = [
      {
        field: question.field,
        value: interpretedValue,
        source: "user",
        confidence,
        confirmed: false,
      },
    ];

    return {
      answer: {
        questionId: question.id,
        field: question.field,
        rawAnswer,
        interpretedValue,
        confidence,
        confirmed: false,
      },
      extractedData,
      inconsistencies: [],
      clarificationRequired: false,
    };
  }

  validateAgainstState(
    result: FinancialInterviewResult,
    state: FinancialIntakeState
  ): FinancialInconsistency[] {
    const inconsistencies: FinancialInconsistency[] = [];

    const value = result.answer.interpretedValue;

    if (
      typeof value === "number" &&
      value < 0 &&
      this.isNonNegativeField(
        result.answer.field
      )
    ) {
      inconsistencies.push({
        id: `negative-${result.answer.field}`,
        fields: [result.answer.field],
        description:
          `El valor indicado para ${result.answer.field} no puede interpretarse como un valor negativo en este contexto.`,
        severity: "medium",
        requiresConfirmation: true,
      });
    }

    const previousValue =
      state.extractedData.find(
        (item) =>
          item.field === result.answer.field
      );

    if (
      previousValue &&
      typeof previousValue.value === "number" &&
      typeof value === "number" &&
      previousValue.value !== value
    ) {
      inconsistencies.push({
        id: `changed-${result.answer.field}`,
        fields: [result.answer.field],
        description:
          `El valor proporcionado anteriormente para ${result.answer.field} es diferente al nuevo valor.`,
        severity: "high",
        requiresConfirmation: true,
      });
    }

    return inconsistencies;
  }

  decideContinuation(
    state: FinancialIntakeState
  ): "continue" | "confirm" | "complete" {
    if (
      state.inconsistencies.some(
        (item) => item.requiresConfirmation
      )
    ) {
      return "confirm";
    }

    if (
      state.missingFields.length === 0 &&
      state.informationStatus === "sufficient"
    ) {
      return "complete";
    }

    return "continue";
  }

  private parseValue(
    question: FinancialIntakeQuestion,
    answer: string
  ): unknown {
    switch (question.type) {
      case "currency":
      case "number":
      case "percentage":
        return this.parseNumber(answer);

      case "boolean":
        return this.parseBoolean(answer);

      case "date":
        return this.parseDate(answer);

      case "choice":
      case "multi_choice":
      case "text":
        return answer;

      default:
        return undefined;
    }
  }

  private parseNumber(
    value: string
  ): number | undefined {
    const normalized = value
      .replace(/\$/g, "")
      .replace(/MXN/gi, "")
      .replace(/USD/gi, "")
      .replace(/,/g, "")
      .replace(/%/g, "")
      .trim();

    const number = Number(normalized);

    return Number.isFinite(number)
      ? number
      : undefined;
  }

  private parseBoolean(
    value: string
  ): boolean | undefined {
    const normalized = value
      .trim()
      .toLowerCase();

    if (
      ["sÃ­", "si", "yes", "true", "1"].includes(
        normalized
      )
    ) {
      return true;
    }

    if (
      ["no", "false", "0"].includes(
        normalized
      )
    ) {
      return false;
    }

    return undefined;
  }

  private parseDate(
    value: string
  ): string | undefined {
    const timestamp = Date.parse(value);

    if (Number.isNaN(timestamp)) {
      return undefined;
    }

    return new Date(timestamp)
      .toISOString()
      .split("T")[0];
  }

  private calculateConfidence(
    question: FinancialIntakeQuestion,
    rawAnswer: string,
    interpretedValue: unknown
  ): number {
    if (!rawAnswer.trim()) {
      return 0;
    }

    if (
      question.type === "currency" ||
      question.type === "number" ||
      question.type === "percentage"
    ) {
      return typeof interpretedValue === "number"
        ? 0.95
        : 0;
    }

    if (
      question.type === "boolean" ||
      question.type === "date"
    ) {
      return interpretedValue !== undefined
        ? 0.95
        : 0;
    }

    return 0.85;
  }

  private buildClarificationQuestion(

      case "date":
        return "Â¿Puedes indicar la fecha o perÃ­odo con mayor precisiÃ³n?";

      case "boolean":
        return "Â¿La respuesta es sÃ­ o no?";

      default:
        return `Â¿Puedes aclarar tu respuesta a esta pregunta: "${question.question}"?`;
    }
  }

  private isNonNegativeField(

    return [
